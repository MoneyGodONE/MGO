const fs = require('fs'); const fetch = require('node-fetch'); const crypto = require('crypto'); const ORG = "MoneyGodONE"; const REPO = "MGO"; const FRONTEND_PATH = "./frontend/index.html"; const REFRESH_INTERVAL = 10 * 60 * 1000; // every 10 minutes async function fetchReadme() { const url = `https://raw.githubusercontent.com/${ORG}/${REPO}/main/README.md`; const res = await fetch(url); if (!res.ok) return "README not found."; return await res.text(); } async function fetchRepos() { const url = `https://api.github.com/users/${ORG}/repos?per_page=100`; const res = await fetch(url); if (!res.ok) throw new Error(`GitHub API error: ${res.status} ${res.statusText}`); return await res.json(); } function hashContent(content) { return crypto.createHash("sha256").update(content).digest("hex"); } function generateHTML(readmeText, repos) { repos.sort((a, b) => new Date(b.updated_at) - new Date(a.updated_at)); const repoCards = repos.map(r => `

${r.name}

${r.description || "No description provided."}

⭐ ${r.stargazers_count} | 🍴 ${r.forks_count}

`).join(''); return ` ${ORG} — Ecosystem

MoneyGod.ONE Ecosystem

Explore the open-source projects powering the MoneyGod.ONE network

📘 NOTE (from MGO README)

${readmeText}

💎 Available Catalogues / Projects

${repoCards}
© ${new Date().getFullYear()} MoneyGod.ONE | Auto-updated every 10 min
`; } async function buildLandingPage() { try { console.log("🔄 Checking for updates..."); const [readmeText, repos] = await Promise.all([fetchReadme(), fetchRepos()]); const html = generateHTML(readmeText, repos); // Check if content changed (compare hash) const newHash = hashContent(html); const oldHash = fs.existsSync(FRONTEND_PATH) ? hashContent(fs.readFileSync(FRONTEND_PATH, "utf8")) : null; if (newHash !== oldHash) { fs.mkdirSync('./frontend', { recursive: true }); fs.writeFileSync(FRONTEND_PATH, html, 'utf8'); console.log(`✅ Updated ${FRONTEND_PATH} with ${repos.length} repos.`); } else { console.log("✅ No changes detected, skipping write."); } } catch (err) { console.error("❌ Error generating landing page:", err.message); } } buildLandingPage(); // 5️⃣ Auto-refresh every 10 minutes setInterval(buildLandingPage, REFRESH_INTERVAL);
